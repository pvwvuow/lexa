    24→    24→function lessonProgressOf(course: Course, progress: Record<string, { status?: string; sectionsSeen?: number; quizBest?: number }>) {
    25→    25→  let completed = 0;
    26→    26→  const flat = course.chapters.flatMap((c) => c.lessons);
    27→    27→  flat.forEach((l) => {
    28→    28→    const p = progress[l.id];
    29→    29→    if (p?.status === "completed") completed += 1;
    30→    30→    else if (p) completed += 0.5;
    31→    31→  });
    32→    32→  return pct(completed, Math.max(1, flat.length));
    33→    33→}

    50→    50→function SectionSlider({
    51→    51→  icon: Icon, title, hint, action, ariaLabel, children, padEnds = false,
    52→    52→}: {
    53→    53→  icon: React.ComponentType<{ className?: string }>;
    54→    54→  title: string;
    55→    55→  hint?: string;

    58→    58→  children: React.ReactNode;
    59→    59→  /** بادسازی اولین و آخرین کارت لبه به لبه در نمایشگر کوچک */
    60→    60→  padEnds?: boolean;
    61→    61→}) {
    62→    62→  const ref = React.useRef<HTMLDivElement>(null);
    63→    63→  const [edge, setEdge] = React.useState<{ prev: boolean; next: boolean }>({ prev: false, next: false });
    64→    64→
    65→    65→  const measure = React.useCallback(() => {
    66→    66→    const el = ref.current;
    67→    67→    if (!el) return;
    68→    68→    const max = el.scrollWidth - el.clientWidth;
    69→    69→    if (max <= 4) return setEdge({ prev: false, next: false });
    70→    70→    const d = Math.abs(el.scrollLeft);
    71→    71→    setEdge({ prev: d > 4, next: max - d > 4 });
    72→    72→  }, []);
    73→    73→
    74→    74→  React.useEffect(() => {
    75→    75→    measure();
    76→    76→    const el = ref.current;
    77→    77→    if (!el) return;
    78→    78→    el.addEventListener("scroll", measure, { passive: true });
    79→    79→    window.addEventListener("resize", measure);
    80→    80→    const t = setTimeout(measure, 400); // پس از نشست فونت‌ها
    81→    81→    return () => {
    82→    82→      el.removeEventListener("scroll", measure);
    83→    83→      window.removeEventListener("resize", measure);
    84→    84→      clearTimeout(t);
    85→    85→    };
    86→    86→  }, [measure]);
    87→    87→
    88→    88→  function slide(dir: 1 | -1) {
    89→    89→    const el = ref.current;
    90→    90→    if (!el) return;
    91→    91→    el.scrollBy({ left: dir * Math.round(el.clientWidth * 0.82), behavior: "smooth" });
    92→    92→  }
    93→    93→
    94→    94→  const prevDisabled = !edge.prev;
    95→    95→  const nextDisabled = !edge.next;
    96→    96→
    97→    97→  return (
    98→    98→    <section className="space-y-3.5" aria-label={ariaLabel}>
    99→    99→      <div className="flex flex-wrap items-center justify-between gap-x-3 gap-y-1.5">
   100→   100→        <h2 className="flex min-w-0 items-center gap-2 text-lg font-bold">
   101→   101→          <Icon className="h-5 w-5 shrink-0 text-bronze" />
   102→   102→          {title}
   103→   103→          {hint && <span className="hidden truncate text-[11px] font-medium text-muted-foreground sm:inline">· {hint}</span>}
   104→   104→        </h2>
   105→   105→        <div className="flex items-center gap-1.5">
   106→   106→          {action}
   107→   107→          <button
   108→   108→            onClick={() => slide(1)}
   109→   109→            disabled={prevDisabled}
   110→   110→            aria-label={`${title} — قبلی`}
   111→   111→            className="grid h-8 w-8 shrink-0 place-items-center rounded-full border border-border bg-card text-muted-foreground shadow-card transition-all hover:border-bronze/60 hover:text-bronze disabled:pointer-events-none disabled:opacity-30"
   112→   112→          >
   113→   113→            <ChevronRight className="h-4 w-4" />
   114→   114→          </button>
   115→   115→          <button
   116→   116→            onClick={() => slide(-1)}
   117→   117→            disabled={nextDisabled}
   118→   118→            aria-label={`${title} — بعدی`}
   119→   119→            className="grid h-8 w-8 shrink-0 place-items-center rounded-full border border-border bg-card text-muted-foreground shadow-card transition-all hover:border-bronze/60 hover:text-bronze disabled:pointer-events-none disabled:opacity-30"
   120→   120→          >
   121→   121→            <ChevronLeft className="h-4 w-4" />
   122→   122→          </button>
   123→   123→        </div>
   124→   124→      </div>
   125→   125→      <div
   126→   126→        ref={ref}
   127→   127→        className={`hslider flex gap-3 overflow-x-auto pb-2 ${
   128→   128→          padEnds ? "-mx-4 px-4 sm:-mx-6 sm:px-6" : ""
   129→   129→        }`}
   130→   130→      >
   131→   131→        {children}
   132→   132→      </div>
   133→   133→    </section>
   134→   134→  );
   135→   135→}
