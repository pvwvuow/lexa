    98→/** تشخیص جهت اسکرول — اسکرول به پایین، نوار بالا/پایین را در موبایل پنهان می‌کند و اسکرول به بالا برمی‌گرداند */
    99→function useHideOnScroll(resetKey: unknown) {
   100→  const [hidden, setHidden] = React.useState(false);
   101→  React.useEffect(() => {
   102→    setHidden(false); // با هر جابه‌جایی مسیر، نوارها دیده می‌شوند
   103→  }, [resetKey]);
   104→  React.useEffect(() => {
   105→    let lastY = window.scrollY;
   106→    let ticking = false;
   107→    const onScroll = () => {
   108→      if (ticking) return;
   109→      ticking = true;
   110→      requestAnimationFrame(() => {
   111→        const y = window.scrollY;
   112→        const delta = y - lastY;
   113→        lastY = y;
   114→        if (delta > 6 && y > 90) setHidden(true);
   115→        else if (delta < -6 || y <= 8) setHidden(false);
   116→        ticking = false;
   117→      });
   118→    };
   119→    window.addEventListener("scroll", onScroll, { passive: true });
   120→    return () => window.removeEventListener("scroll", onScroll);
   121→  }, []);
   122→  return hidden;
   123→}

   125→function lessonPctOf(course: Course, progress: Record<string, { status?: string }>) {
   126→  const flat = course.chapters.flatMap((c) => c.lessons);
   127→  let done = 0;
   128→  flat.forEach((l) => {
   129→    if (progress[l.id]?.status === "completed") done += 1;
   130→    else if (progress[l.id]) done += 0.5;
   131→  });
   132→  return pct(done, Math.max(1, flat.length));
   133→}
