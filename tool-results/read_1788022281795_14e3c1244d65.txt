    99→/** تشخیص جهت اسکرول — اسکرول به پایین، نوار بالا/پایین را در موبایل پنهان می‌کند و اسکرول به بالا برمی‌گرداند */
   100→function useHideOnScroll(resetKey: unknown) {
   101→  const [hidden, setHidden] = React.useState(false);
   102→  React.useEffect(() => {
   103→    setHidden(false); // با هر جابه‌جایی مسیر، نوارها دیده می‌شوند
   104→  }, [resetKey]);
   105→  React.useEffect(() => {
   106→    let lastY = window.scrollY;
   107→    let ticking = false;
   108→    const onScroll = () => {
   109→      if (ticking) return;
   110→      ticking = true;
   111→      requestAnimationFrame(() => {
   112→        const y = window.scrollY;
   113→        const delta = y - lastY;
   114→        lastY = y;
   115→        if (delta > 6 && y > 90) setHidden(true);
   116→        else if (delta < -6 || y <= 8) setHidden(false);
   117→        ticking = false;
   118→      });
   119→    };
   120→    window.addEventListener("scroll", onScroll, { passive: true });
   121→    return () => window.removeEventListener("scroll", onScroll);
   122→  }, []);
   123→  return hidden;
   124→}

   126→function lessonPctOf(course: Course, progress: Record<string, { status?: string }>) {
   127→  const flat = course.chapters.flatMap((c) => c.lessons);
   128→  let done = 0;
   129→  flat.forEach((l) => {
   130→    if (progress[l.id]?.status === "completed") done += 1;
   131→    else if (progress[l.id]) done += 0.5;
   132→  });
   133→  return pct(done, Math.max(1, flat.length));
   134→}
