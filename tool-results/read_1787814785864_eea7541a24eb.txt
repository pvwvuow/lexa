    56→    56→function prettyLesson(id: string): string {
    57→    57→  const m = id.match(/^(m|t|md7)-l(\d+)-(\d+)$/);
    58→    58→  if (!m) return id;
    59→    59→  const book = m[1] === "t" ? "تجارت ۳" : m[1] === "md7" ? "مدنی ۷" : `مدنی ${m[1] === "m" ? "" : ""}${m[2]}`;
    60→    60→  return `${book} — جلسهٔ ${fa(m[3])} فصل ${fa(m[2])}`;
    61→    61→}
