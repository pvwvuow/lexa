    24→function lessonProgressOf(course: Course, progress: Record<string, { status?: string; sectionsSeen?: number; quizBest?: number }>) {
    25→  let completed = 0;
    26→  const flat = course.chapters.flatMap((c) => c.lessons);
    27→  flat.forEach((l) => {
    28→    const p = progress[l.id];
    29→    if (p?.status === "completed") completed += 1;
    30→    else if (p) completed += 0.5;
    31→  });
    32→  return pct(completed, Math.max(1, flat.length));
    33→}

    35→/** تعداد جلسات تمام‌شدهٔ یک درس */
    36→function doneCountOf(course: Course, progress: Record<string, { status?: string }>) {
    37→  return course.chapters.reduce((n, ch) => n + ch.lessons.filter((l) => progress[l.id]?.status === "completed").length, 0);
    38→}

    41→function nextPending(course: Course, progress: Record<string, { status?: string }>): Lesson | null {
    42→  for (const ch of [...course.chapters].sort((a, b) => a.order - b.order))
    43→    for (const l of ch.lessons)
    44→      if (!progress[l.id]?.status) return l;
    45→  return null;
    46→}

    50→function SectionSlider({
    51→  icon: Icon, title, hint, action, ariaLabel, children, padEnds = false,
    52→}: {
    53→  icon: React.ComponentType<{ className?: string }>;
    54→  title: string;
    55→  hint?: string;
    56→  action?: React.ReactNode;
    57→  ariaLabel: string;
    58→  children: React.ReactNode;
    59→  /** بادسازی اولین و آخرین کارت لبه به لبه در نمایشگر کوچک */
    60→  padEnds?: boolean;
    61→}) {
    62→  const ref = React.useRef<HTMLDivElement>(null);
    63→  const [edge, setEdge] = React.useState<{ prev: boolean; next: boolean }>({ prev: false, next: false });
    64→
    65→  const measure = React.useCallback(() => {
    66→    const el = ref.current;
    67→    if (!el) return;
    68→    const max = el.scrollWidth - el.clientWidth;
    69→    if (max <= 4) return setEdge({ prev: false, next: false });
    70→    const d = Math.abs(el.scrollLeft);
    71→    setEdge({ prev: d > 4, next: max - d > 4 });
    72→  }, []);
    73→
    74→  React.useEffect(() => {
    75→    measure();
    76→    const el = ref.current;
    77→    if (!el) return;
    78→    el.addEventListener("scroll", measure, { passive: true });
    79→    window.addEventListener("resize", measure);
    80→    const t = setTimeout(measure, 400); // پس از نشست فونت‌ها
    81→    return () => {
    82→      el.removeEventListener("scroll", measure);
    83→      window.removeEventListener("resize", measure);
    84→      clearTimeout(t);
    85→    };
    86→  }, [measure]);
    87→
    88→  function slide(dir: 1 | -1) {
    89→    const el = ref.current;
    90→    if (!el) return;
    91→    el.scrollBy({ left: dir * Math.round(el.clientWidth * 0.82), behavior: "smooth" });
    92→  }
    93→
    94→  const prevDisabled = !edge.prev;
    95→  const nextDisabled = !edge.next;
    96→
    97→  return (
    98→    <section className="space-y-3.5" aria-label={ariaLabel}>
    99→      <div className="flex flex-wrap items-center justify-between gap-x-3 gap-y-1.5">
   100→        <h2 className="flex min-w-0 items-center gap-2 text-lg font-bold">
   101→          <Icon className="h-5 w-5 shrink-0 text-bronze" />
   102→          {title}
   103→          {hint && <span className="hidden truncate text-[11px] font-medium text-muted-foreground sm:inline">· {hint}</span>}
   104→        </h2>
   105→        <div className="flex items-center gap-1.5">
   106→          {action}
   107→          <button
   108→            onClick={() => slide(1)}
   109→            disabled={prevDisabled}
   110→            aria-label={`${title} — قبلی`}
   111→            className="grid h-8 w-8 shrink-0 place-items-center rounded-full border border-border bg-card text-muted-foreground shadow-card transition-all hover:border-bronze/60 hover:text-bronze disabled:pointer-events-none disabled:opacity-30"
   112→          >
   113→            <ChevronRight className="h-4 w-4" />
   114→          </button>
   115→          <button
   116→            onClick={() => slide(-1)}
   117→            disabled={nextDisabled}
   118→            aria-label={`${title} — بعدی`}
   119→            className="grid h-8 w-8 shrink-0 place-items-center rounded-full border border-border bg-card text-muted-foreground shadow-card transition-all hover:border-bronze/60 hover:text-bronze disabled:pointer-events-none disabled:opacity-30"
   120→          >
   121→            <ChevronLeft className="h-4 w-4" />
   122→          </button>
   123→        </div>
   124→      </div>
   125→      <div
   126→        ref={ref}
   127→        className={`hslider flex gap-3 overflow-x-auto pb-2 ${
   128→          padEnds ? "-mx-4 px-4 sm:-mx-6 sm:px-6" : ""
   129→        }`}
   130→      >
   131→        {children}
   132→      </div>
   133→    </section>
   134→  );
   135→}

   302→function MyLibraryShelf({
   303→  courses, progress,

   305→  courses: Course[];
   306→  progress: Record<string, { status?: string }>;
   307→}) {
   308→  const tBooks = useApp((s) => s.tBooks);

   311→  const teacherCount = courses.filter((c) => !!((c as Course & { _ownerUsername?: string })._ownerUsername)).length;
   312→
   313→  /** حذف دورهٔ استاد از کتابخانه — رکورد سرور پاک می‌شود ولی هیچ دادهٔ یادگیری دست نمی‌خورد */
   314→  async function removeTeacherBook(courseId: string, title: string) {
   315→    try {
   316→      await fetch("/api/library", {
   317→        method: "POST",
   318→        headers: { "Content-Type": "application/json" },
   319→        body: JSON.stringify({ courseId }),
   320→      });
   321→      await refreshLibrary();

   324→  }

   330→      willHide

   334→  }

   337→    <button
   338→      onClick={() => navigate({ view: "library" })}
   339→      className="group flex w-[164px] shrink-0 snap-start flex-col items-center justify-center gap-2 rounded-2xl border border-dashed border-border/90 bg-card/60 p-4 text-center transition-colors hover:border-bronze/50 sm:w-[184px]"

   351→    <SectionSlider

   355→      ariaLabel="قفسهٔ کتابخانهٔ من"
   356→      padEnds
   357→      action={
   358→        <span className="hidden text-[11px] font-medium text-muted-foreground sm:inline">
   359→          طول کشید؟ خودش می‌غزد
   360→        </span>
   361→      }
   362→    >
   363→      {courses.length === 0 ? (
   364→        <button
   365→          onClick={() => navigate({ view: "library" })}
   366→          className="w-full shrink-0 rounded-2xl border border-dashed border-border bg-card px-5 py-6 text-start text-sm text-muted-foreground transition-colors hover:border-bronze/50"
   367→        >
   368→          هنوز درسی در قفسه نداری؛ از «کتابخانهٔ عمومی» یکی را انتخاب کن تا همین‌جا ببینی‌اش.
   369→        </button>
   370→      ) : (
   371→        courses.map((c) => {
   372→          const cid = c.id;
   373→          const ownerCourse =
   374→            !!((c as Course & { _ownerUsername?: string })._ownerUsername) &&
   375→            tBooks.some((t) => t.id === cid);
   376→          const isBuiltin = BUILTIN_IDS.has(cid);

   378→            <BookCover
   379→              key={cid}
   380→              c={c}
   381→              progress={progress}
   382→              onRemove={
   383→                ownerCourse
   384→                  ? () => removeTeacherBook(cid, c.title)
   385→                  : isBuiltin
   386→                    ? () => toggleBuiltin(cid, c.title)
   387→                    : undefined
   388→              }
   389→            />
   390→          );
   391→        })
   392→      )}
   393→      {addGhostCard}
   394→    </SectionSlider>
   395→  );
   396→}
