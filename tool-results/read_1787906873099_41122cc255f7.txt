    26→function lessonProgressOf(course: Course, progress: Record<string, { status?: string; sectionsSeen?: number; quizBest?: number }>) {
    27→  let completed = 0;
    28→  const flat = course.chapters.flatMap((c) => c.lessons);
    29→  flat.forEach((l) => {
    30→    const p = progress[l.id];
    31→    if (p?.status === "completed") completed += 1;
    32→    else if (p) completed += 0.5;
    33→  });
    34→  return Math.round((completed / Math.max(1, flat.length)) * 100);
    35→}
    36→
    37→/** تعداد جلسات تمام‌شدهٔ یک درس */
    38→function doneCountOf(course: Course, progress: Record<string, { status?: string }>) {
    39→  return course.chapters.reduce((n, ch) => n + ch.lessons.filter((l) => progress[l.id]?.status === "completed").length, 0);
    40→}

    42→/** اولین جلسهٔ ناتمام درس (برای «ادامه از …») */
    43→function nextPending(course: Course, progress: Record<string, { status?: string }>): Lesson | null {
    44→  for (const ch of [...course.chapters].sort((a, b) => a.order - b.order))
    45→    for (const l of ch.lessons)
    46→      if (!progress[l.id]?.status) return l;
    47→  return null;
    48→}

    52→function SectionSlider({
    53→  icon: Icon, title, hint, action, ariaLabel, children, padEnds = false,
    54→}: {
    55→  icon: React.ComponentType<{ className?: string }>;
    56→  title: string;
    57→  hint?: string;
    58→  action?: React.ReactNode;
    59→  ariaLabel: string;
    60→  children: React.ReactNode;
    61→  /** بادسازی اولین و آخرین کارت لبه به لبه در نمایشگر کوچک */
    62→  padEnds?: boolean;
    63→}) {
    64→  const ref = React.useRef<HTMLDivElement>(null);
    65→  const [edge, setEdge] = React.useState<{ prev: boolean; next: boolean }>({ prev: false, next: false });
    66→
    67→  const measure = React.useCallback(() => {
    68→    const el = ref.current;
    69→    if (!el) return;
    70→    const max = el.scrollWidth - el.clientWidth;
    71→    if (max <= 4) return setEdge({ prev: false, next: false });
    72→    const d = Math.abs(el.scrollLeft);
    73→    setEdge({ prev: d > 4, next: max - d > 4 });
    74→  }, []);
    75→
    76→  React.useEffect(() => {
    77→    measure();
    78→    const el = ref.current;
    79→    if (!el) return;
    80→    el.addEventListener("scroll", measure, { passive: true });
    81→    window.addEventListener("resize", measure);
    82→    const t = setTimeout(measure, 400); // پس از نشست فونت‌ها
    83→    return () => {
    84→      el.removeEventListener("scroll", measure);
    85→      window.removeEventListener("resize", measure);
    86→      clearTimeout(t);
    87→    };
    88→  }, [measure]);
    89→
    90→  function slide(dir: 1 | -1) {
    91→    const el = ref.current;
    92→    if (!el) return;
    93→    el.scrollBy({ left: dir * Math.round(el.clientWidth * 0.82), behavior: "smooth" });
    94→  }
    95→
    96→  const prevDisabled = !edge.prev;
    97→  const nextDisabled = !edge.next;
    98→
    99→  return (
   100→    <section className="space-y-3.5" aria-label={ariaLabel}>
   101→      <div className="flex flex-wrap items-center justify-between gap-x-3 gap-y-1.5">
   102→        <h2 className="flex min-w-0 items-center gap-2 text-lg font-bold">
   103→          <Icon className="h-5 w-5 shrink-0 text-bronze" />
   104→          {title}
   105→          {hint && <span className="hidden truncate text-[11px] font-medium text-muted-foreground sm:inline">· {hint}</span>}
   106→        </h2>
   107→        <div className="flex items-center gap-1.5">
   108→          {action}
   109→          <button
   110→            onClick={() => slide(1)}
   111→            disabled={prevDisabled}
   112→            aria-label={`${title} — قبلی`}
   113→            className="grid h-8 w-8 shrink-0 place-items-center rounded-full border border-border bg-card text-muted-foreground shadow-card transition-all hover:border-bronze/60 hover:text-bronze disabled:pointer-events-none disabled:opacity-30"
   114→          >
   115→            <ChevronRight className="h-4 w-4" />
   116→          </button>
   117→          <button
   118→            onClick={() => slide(-1)}
   119→            disabled={nextDisabled}
   120→            aria-label={`${title} — بعدی`}
   121→            className="grid h-8 w-8 shrink-0 place-items-center rounded-full border border-border bg-card text-muted-foreground shadow-card transition-all hover:border-bronze/60 hover:text-bronze disabled:pointer-events-none disabled:opacity-30"
   122→          >
   123→            <ChevronLeft className="h-4 w-4" />
   124→          </button>
   125→        </div>
   126→      </div>
   127→      <div
   128→        ref={ref}
   129→        className={`hslider flex gap-3 overflow-x-auto pb-2 ${
   130→          padEnds ? "-mx-4 px-4 sm:-mx-6 sm:px-6" : ""
   131→        }`}
   132→      >
   133→        {children}
   134→      </div>
   135→    </section>
   136→  );
   137→}
