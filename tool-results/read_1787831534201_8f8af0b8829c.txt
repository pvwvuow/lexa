    21→function lessonProgressOf(course: Course, progress: Record<string, { status?: string; sectionsSeen?: number; quizBest?: number }>) {
    22→  let completed = 0;
    23→  const flat = course.chapters.flatMap((c) => c.lessons);
    24→  flat.forEach((l) => {
    25→    const p = progress[l.id];
    26→    if (p?.status === "completed") completed += 1;
    27→    else if (p) completed += 0.5;
    28→  });
    29→  return pct(completed, Math.max(1, flat.length));
    30→}
    31→
    32→/** تعداد جلسات تمام‌شدهٔ یک درس */
    33→function doneCountOf(course: Course, progress: Record<string, { status?: string }>) {
    34→  return course.chapters.reduce((n, ch) => n + ch.lessons.filter((l) => progress[l.id]?.status === "completed").length, 0);
    35→}

    37→/** اولین جلسهٔ ناتمام درس (برای «ادامه از …») */
    38→function nextPending(course: Course, progress: Record<string, { status?: string }>): Lesson | null {
    39→  for (const ch of [...course.chapters].sort((a, b) => a.order - b.order))
    40→    for (const l of ch.lessons)
    41→      if (!progress[l.id]?.status) return l;
    42→  return null;
    43→}

   335→        method: "POST",
   336→        headers: { "Content-Type": "application/json" },
   337→        body: JSON.stringify({ courseId }),
   338→      });
   339→      await refreshLibrary();

   357→        </div>

   360→      {/* قفسهٔ کتاب‌ها */}
   361→      {removedMsg && (
