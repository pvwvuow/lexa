    25→function lessonProgressOf(course: Course, progress: Record<string, { status?: string; sectionsSeen?: number; quizBest?: number }>) {
    26→  let completed = 0;
    27→  const flat = course.chapters.flatMap((c) => c.lessons);
    28→  flat.forEach((l) => {
    29→    const p = progress[l.id];
    30→    if (p?.status === "completed") completed += 1;
    31→    else if (p) completed += 0.5;
    32→  });
    33→  return Math.round((completed / Math.max(1, flat.length)) * 100);
    34→}
    35→
    36→/** تعداد جلسات تمام‌شدهٔ یک درس */
    37→function doneCountOf(course: Course, progress: Record<string, { status?: string }>) {
    38→  return course.chapters.reduce((n, ch) => n + ch.lessons.filter((l) => progress[l.id]?.status === "completed").length, 0);
    39→}

    41→/** اولین جلسهٔ ناتمام درس (برای «ادامه از …») */
    42→function nextPending(course: Course, progress: Record<string, { status?: string }>): Lesson | null {
    43→  for (const ch of [...course.chapters].sort((a, b) => a.order - b.order))
    44→    for (const l of ch.lessons)
    45→      if (!progress[l.id]?.status) return l;
    46→  return null;
    47→}

    51→function SectionSlider({
    52→  icon: Icon, title, hint, action, ariaLabel, children, padEnds = false,
    53→}: {
    54→  icon: React.ComponentType<{ className?: string }>;
    55→  title: string;
    56→  hint?: string;
    57→  action?: React.ReactNode;
    58→  ariaLabel: string;
    59→  children: React.ReactNode;
    60→  /** بادسازی اولین و آخرین کارت لبه به لبه در نمایشگر کوچک */
    61→  padEnds?: boolean;
    62→}) {
    63→  const ref = React.useRef<HTMLDivElement>(null);
    64→  const [edge, setEdge] = React.useState<{ prev: boolean; next: boolean }>({ prev: false, next: false });
    65→
    66→  const measure = React.useCallback(() => {
    67→    const el = ref.current;
    68→    if (!el) return;
    69→    const max = el.scrollWidth - el.clientWidth;
    70→    if (max <= 4) return setEdge({ prev: false, next: false });
    71→    const d = Math.abs(el.scrollLeft);
    72→    setEdge({ prev: d > 4, next: max - d > 4 });
    73→  }, []);
    74→
    75→  React.useEffect(() => {
    76→    measure();
    77→    const el = ref.current;
    78→    if (!el) return;
    79→    el.addEventListener("scroll", measure, { passive: true });
    80→    window.addEventListener("resize", measure);
    81→    const t = setTimeout(measure, 400); // پس از نشست فونت‌ها
    82→    return () => {
    83→      el.removeEventListener("scroll", measure);
    84→      window.removeEventListener("resize", measure);
    85→      clearTimeout(t);
    86→    };
    87→  }, [measure]);
    88→
    89→  function slide(dir: 1 | -1) {
    90→    const el = ref.current;
    91→    if (!el) return;
    92→    el.scrollBy({ left: dir * Math.round(el.clientWidth * 0.82), behavior: "smooth" });
    93→  }
    94→
    95→  const prevDisabled = !edge.prev;
    96→  const nextDisabled = !edge.next;
    97→
    98→  return (
    99→    <section className="space-y-3.5" aria-label={ariaLabel}>
   100→      <div className="flex flex-wrap items-center justify-between gap-x-3 gap-y-1.5">
   101→        <h2 className="flex min-w-0 items-center gap-2 text-lg font-bold">
   102→          <Icon className="h-5 w-5 shrink-0 text-bronze" />
   103→          {title}
   104→          {hint && <span className="hidden truncate text-[11px] font-medium text-muted-foreground sm:inline">· {hint}</span>}
   105→        </h2>
   106→        <div className="flex items-center gap-1.5">
   107→          {action}
   108→          <button
   109→            onClick={() => slide(1)}
   110→            disabled={prevDisabled}
   111→            aria-label={`${title} — قبلی`}
   112→            className="grid h-8 w-8 shrink-0 place-items-center rounded-full border border-border bg-card text-muted-foreground shadow-card transition-all hover:border-bronze/60 hover:text-bronze disabled:pointer-events-none disabled:opacity-30"
   113→          >
   114→            <ChevronRight className="h-4 w-4" />
   115→          </button>
   116→          <button
   117→            onClick={() => slide(-1)}
   118→            disabled={nextDisabled}
   119→            aria-label={`${title} — بعدی`}
   120→            className="grid h-8 w-8 shrink-0 place-items-center rounded-full border border-border bg-card text-muted-foreground shadow-card transition-all hover:border-bronze/60 hover:text-bronze disabled:pointer-events-none disabled:opacity-30"
   121→          >
   122→            <ChevronLeft className="h-4 w-4" />
   123→          </button>
   124→        </div>
   125→      </div>
   126→      <div
   127→        ref={ref}
   128→        className={`hslider flex gap-3 overflow-x-auto pb-2 ${
   129→          padEnds ? "-mx-4 px-4 sm:-mx-6 sm:px-6" : ""
   130→        }`}
   131→      >
   132→        {children}
   133→      </div>
   134→    </section>
   135→  );
   136→}
