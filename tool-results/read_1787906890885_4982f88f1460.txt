    32→  switch (type) {
    33→    case "law":
    34→      return { id, type: "law", law: [{ no: "", source: "", text: "" }] };

    36→      return { id, type: "notes", bullets: [""] };
    37→    case "compare":

    42→      return { id, type: "example", body: "" };

   235→/* ═══ دیالوگ عمومی ═══════════════════════════════════════════════════════════ */
   236→function Modal({ open, onClose, children, wide }: { open: boolean; onClose: () => void; children: React.ReactNode; wide?: boolean }) {
   237→  if (!open) return null;
   238→  return (
   239→    <div className="fixed inset-0 z-[70] grid place-items-start justify-items-center overflow-y-auto bg-black/45 p-3 backdrop-blur-sm sm:p-6" role="dialog" aria-modal="true" onClick={onClose}>
   240→      <div
   241→        onClick={(e) => e.stopPropagation()}
   242→        className={`w-full ${wide ? "max-w-3xl" : "max-w-xl"} rounded-2xl border border-border bg-card p-5 shadow-2xl sm:p-6 my-4 max-h-none`}
   243→      >
   244→        {children}
   245→      </div>
   246→    </div>
   247→  );
   248→}

   251→interface PostDraft { id?: string; title: string; summary: string; tags: string; category: string; categories: string[]; blocks: LessonSection[]; thumbnail: string; quiz: QuizDraft[] }
   252→const EMPTY_POST: PostDraft = { title: "", summary: "", tags: "", category: "", categories: [], blocks: [], thumbnail: "", quiz: [] };

   254→/** چیپ‌های انتخاب چندشاخه — استاد می‌تواند هر چند گزینه که خواست انتخاب کند */
   255→function CategoryChips({ value, onChange }: { value: string[]; onChange: (next: string[]) => void }) {
   256→  return (
   257→    <div className="flex flex-wrap gap-1.5">
   258→      {CATEGORIES.map((c) => {

   260→        return (
   261→          <button
   262→            key={c.slug}
   263→            type="button"
   264→            onClick={() => onChange(on ? value.filter((v) => v !== c.slug) : [...value, c.slug])}
   265→            aria-pressed={on}
   266→            title={c.desc}
   267→            className={`rounded-full border px-3.5 py-1.5 text-[11.5px] font-bold transition-all ${
   268→              on
   269→                ? "border-bronze bg-bronze/15 text-bronze shadow-card"
   270→                : "border-border bg-background text-muted-foreground hover:border-bronze/40 hover:text-foreground"

   272→          >
   273→            {on ? "✓ " : ""}{c.label}
   274→          </button>
   275→        );
   276→      })}
   277→    </div>
   278→  );

   281→export function PostEditor({ draft, onClose, onSaved }: { draft: PostDraft | null; onClose: () => void; onSaved: () => void }) {
   282→  const [d, setD] = React.useState<PostDraft>(draft ?? EMPTY_POST);
   283→  const [busy, setBusy] = React.useState(false);
   284→  const [err, setErr] = React.useState("");
   285→
   286→  React.useEffect(() => setD(draft ?? EMPTY_POST), [draft]);
   287→
   288→  async function save() {
   289→    setBusy(true); setErr("");
   290→    try {
   291→      const res = await fetch(d.id ? `/api/posts/${d.id}` : "/api/posts", {
   292→        method: d.id ? "PUT" : "POST",
   293→        headers: { "Content-Type": "application/json" },
   294→        body: JSON.stringify({ ...d, quiz: draftToQuizPayload(d.quiz) }),
   295→      });
   296→      const j = await res.json();
   297→      if (!res.ok) throw new Error(j.error ?? "ذخیره ناموفق بود.");
   298→      onSaved();
   299→      onClose();
   300→    } catch (e) {
   301→      setErr(e instanceof Error ? e.message : "خطایی رخ داد.");
   302→    } finally {
   303→      setBusy(false);
   304→    }
   305→  }
   306→
   307→  return (
   308→    <Modal open={!!draft} onClose={onClose} wide>
   309→      <div className="mb-4 flex items-center gap-2.5">
   310→        <span className="grid h-10 w-10 place-items-center rounded-xl bg-primary/10 text-primary"><PenSquare className="h-5 w-5" /></span>
   311→        <h2 className="font-display text-lg font-bold">{d.id ? "ویرایش مطلب" : "مطلب جدید"}</h2>
   312→        <button onClick={onClose} aria-label="بستن" className="ms-auto rounded-lg p-1.5 text-muted-foreground hover:bg-muted"><X className="h-5 w-5" /></button>
   313→      </div>
   314→
   315→      <div className="space-y-3">
   316→        <div>
   317→          <label className={labelCls}>عنوان مطلب *</label>
   318→          <input value={d.title} onChange={(e) => setD({ ...d, title: e.target.value })} placeholder="مثلاً: سه اشتباه رایج در قرارداد بیع" className={inputCls} />
   319→        </div>
   320→        <div className="grid gap-3 sm:grid-cols-2">
   321→          <div>
   322→            <label className={labelCls}>خلاصه (برای فید خانه)</label>
   323→            <input value={d.summary} onChange={(e) => setD({ ...d, summary: e.target.value })} maxLength={280} className={inputCls} />
   324→          </div>
   325→          <div>
   326→            <label className={labelCls}>برچسب‌ها (با کاما)</label>
   327→            <input value={d.tags} onChange={(e) => setD({ ...d, tags: e.target.value })} placeholder="قراردادها، بیع، نکته امتحانی" className={inputCls} />
   328→          </div>
   329→        </div>
   330→        {/* شاخه‌های کتابخانهٔ عمومی — چندگانه؛ مطلب در همهٔ دسته‌های انتخابی دیده می‌شود */}
   331→        <div>
   332→          <label className={labelCls}>شاخه‌های کتابخانهٔ عمومی (چند گزینه مجاز است)</label>
   333→          <CategoryChips
   334→            value={d.categories}
   335→            onChange={(cats) => setD({ ...d, categories: cats, category: cats[0] ?? "" })}
   336→          />
   337→          <p className="mt-1 text-[10.5px] leading-relaxed text-muted-foreground">اگر شاخه انتخاب کنی، این مطلب در هر یک از آن دسته‌ها هم نمایش داده می‌شود؛ بدون انتخاب فقط در فید دیده می‌شود.</p>
   338→        </div>
   339→        {/* تصویر شاخص دلخواه — آپلود یا لینک؛ اگر خالی باشد جلد رنگی خودکار می‌آید */}
   340→        <ThumbnailPicker value={d.thumbnail} onChange={(thumbnail) => setD({ ...d, thumbnail })} />
   341→        <div>
   342→          <label className={labelCls}>محتوا</label>
   343→          <BlockEditor blocks={d.blocks} onChange={(blocks) => setD({ ...d, blocks })} />
   344→        </div>
   345→        {/* آزمون پایان مبحث — دلخواه؛ دانشجو بعد از مطالعه آن را می‌بیند */}
   346→        <QuizEditor
   347→          label="آزمون پایان این مبحث"
   348→          questions={d.quiz}
   349→          onChange={(quiz) => setD({ ...d, quiz })}
   350→        />
   351→        {err && <p className="rounded-lg bg-destructive/10 px-3 py-2 text-sm text-destructive">{err}</p>}
   352→        <div className="flex justify-end gap-2 pt-1">
   353→          <button onClick={onClose} className="rounded-xl border border-border px-4 py-2.5 text-sm font-semibold text-muted-foreground hover:text-foreground">انصراف</button>
   354→          <button onClick={save} disabled={busy || !d.title.trim()} className="inline-flex items-center gap-2 rounded-xl bg-primary px-5 py-2.5 text-sm font-bold text-primary-foreground disabled:opacity-45">
   355→            {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : <Save className="h-4 w-4" />} انتشار / ذخیره
