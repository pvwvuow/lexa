    25→function newBlock(type: string): LessonSection {
    26→  const id = `b-${Math.random().toString(36).slice(2, 9)}`;
    27→  switch (type) {
    28→    case "law":
    29→      return { id, type: "law", law: [{ no: "", source: "", text: "" }] };
    30→    case "notes":
    31→      return { id, type: "notes", bullets: [""] };
    32→    case "compare":
    33→      return { id, type: "compare", table: { headers: ["", ""], rows: [["", ""]] } };
    34→    case "question":
    35→      return { id, type: "question", questionText: "", suggestedAnswer: "" };
    36→    case "example":
    37→      return { id, type: "example", body: "" };
    38→    case "intro":
    39→      return { id, type: "intro", body: "" };
    40→    default:
    41→      return { id, type: "concept", body: "" };
    42→  }

   231→function Modal({ open, onClose, children, wide }: { open: boolean; onClose: () => void; children: React.ReactNode; wide?: boolean }) {
   232→  if (!open) return null;

   234→    <div className="fixed inset-0 z-[70] grid place-items-start justify-items-center overflow-y-auto bg-black/45 p-3 backdrop-blur-sm sm:p-6" role="dialog" aria-modal="true" onClick={onClose}>
   235→      <div
   236→        onClick={(e) => e.stopPropagation()}
   237→        className={`w-full ${wide ? "max-w-3xl" : "max-w-xl"} rounded-2xl border border-border bg-card p-5 shadow-2xl sm:p-6 my-4 max-h-none`}
   238→      >
   239→        {children}
   240→      </div>
   241→    </div>
   242→  );
   243→}

   276→export function PostEditor({ draft, onClose, onSaved }: { draft: PostDraft | null; onClose: () => void; onSaved: () => void }) {
   277→  const [d, setD] = React.useState<PostDraft>(draft ?? EMPTY_POST);
   278→  const [busy, setBusy] = React.useState(false);
   279→  const [err, setErr] = React.useState("");
   280→
   281→  React.useEffect(() => setD(draft ?? EMPTY_POST), [draft]);
   282→
   283→  async function save() {
   284→    setBusy(true); setErr("");
   285→    try {
   286→      const res = await fetch(d.id ? `/api/posts/${d.id}` : "/api/posts", {
   287→        method: d.id ? "PUT" : "POST",
   288→        headers: { "Content-Type": "application/json" },
   289→        body: JSON.stringify(d),
   290→      });
   291→      const j = await res.json();
   292→      if (!res.ok) throw new Error(j.error ?? "ذخیره ناموفق بود.");
   293→      onSaved();
   294→      onClose();
   295→    } catch (e) {
   296→      setErr(e instanceof Error ? e.message : "خطایی رخ داد.");
   297→    } finally {
   298→      setBusy(false);
   299→    }
   300→  }
   301→
   302→  return (
   303→    <Modal open={!!draft} onClose={onClose} wide>
   304→      <div className="mb-4 flex items-center gap-2.5">
   305→        <span className="grid h-10 w-10 place-items-center rounded-xl bg-primary/10 text-primary"><PenSquare className="h-5 w-5" /></span>
   306→        <h2 className="font-display text-lg font-bold">{d.id ? "ویرایش مطلب" : "مطلب جدید"}</h2>
   307→        <button onClick={onClose} aria-label="بستن" className="ms-auto rounded-lg p-1.5 text-muted-foreground hover:bg-muted"><X className="h-5 w-5" /></button>
   308→      </div>
   309→
   310→      <div className="space-y-3">
   311→        <div>
   312→          <label className={labelCls}>عنوان مطلب *</label>
   313→          <input value={d.title} onChange={(e) => setD({ ...d, title: e.target.value })} placeholder="مثلاً: سه اشتباه رایج در قرارداد بیع" className={inputCls} />
   314→        </div>
   315→        <div className="grid gap-3 sm:grid-cols-2">
   316→          <div>
   317→            <label className={labelCls}>خلاصه (برای فید خانه)</label>
   318→            <input value={d.summary} onChange={(e) => setD({ ...d, summary: e.target.value })} maxLength={280} className={inputCls} />
   319→          </div>
   320→          <div>
   321→            <label className={labelCls}>برچسب‌ها (با کاما)</label>
   322→            <input value={d.tags} onChange={(e) => setD({ ...d, tags: e.target.value })} placeholder="قراردادها، بیع، نکته امتحانی" className={inputCls} />
   323→          </div>
   324→        </div>
   325→        {/* شاخه‌های کتابخانهٔ عمومی — چندگانه؛ مطلب در همهٔ دسته‌های انتخابی دیده می‌شود */}
   326→        <div>
   327→          <label className={labelCls}>شاخه‌های کتابخانهٔ عمومی (چند گزینه مجاز است)</label>
   328→          <CategoryChips
   329→            value={d.categories}
   330→            onChange={(cats) => setD({ ...d, categories: cats, category: cats[0] ?? "" })}
   331→          />
   332→          <p className="mt-1 text-[10.5px] leading-relaxed text-muted-foreground">اگر شاخه انتخاب کنی، این مطلب در هر یک از آن دسته‌ها هم نمایش داده می‌شود؛ بدون انتخاب فقط در فید دیده می‌شود.</p>
   333→        </div>
   334→        <div>
   335→          <label className={labelCls}>محتوا</label>
   336→          <BlockEditor blocks={d.blocks} onChange={(blocks) => setD({ ...d, blocks })} />
   337→        </div>
   338→        {err && <p className="rounded-lg bg-destructive/10 px-3 py-2 text-sm text-destructive">{err}</p>}
   339→        <div className="flex justify-end gap-2 pt-1">
   340→          <button onClick={onClose} className="rounded-xl border border-border px-4 py-2.5 text-sm font-semibold text-muted-foreground hover:text-foreground">انصراف</button>
   341→          <button onClick={save} disabled={busy || !d.title.trim()} className="inline-flex items-center gap-2 rounded-xl bg-primary px-5 py-2.5 text-sm font-bold text-primary-foreground disabled:opacity-45">
   342→            {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : <Save className="h-4 w-4" />} انتشار / ذخیره
   343→          </button>
   344→        </div>
   345→      </div>
   346→    </Modal>
   347→  );
   348→}
