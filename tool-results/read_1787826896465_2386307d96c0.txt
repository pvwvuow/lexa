   244→/* ═══ ویرایشگر مطلب ══════════════════════════════════════════════════════════ */
   245→interface PostDraft { id?: string; title: string; summary: string; tags: string; blocks: LessonSection[] }
   246→const EMPTY_POST: PostDraft = { title: "", summary: "", tags: "", blocks: [] };
   247→
   248→export function PostEditor({ draft, onClose, onSaved }: { draft: PostDraft | null; onClose: () => void; onSaved: () => void }) {
   249→  const [d, setD] = React.useState<PostDraft>(draft ?? EMPTY_POST);
   250→  const [busy, setBusy] = React.useState(false);
   251→  const [err, setErr] = React.useState("");
   252→
   253→  React.useEffect(() => setD(draft ?? EMPTY_POST), [draft]);
   254→
   255→  async function save() {
   256→    setBusy(true); setErr("");
   257→    try {
   258→      const res = await fetch(d.id ? `/api/posts/${d.id}` : "/api/posts", {
   259→        method: d.id ? "PUT" : "POST",
   260→        headers: { "Content-Type": "application/json" },
   261→        body: JSON.stringify(d),
   262→      });
   263→      const j = await res.json();
   264→      if (!res.ok) throw new Error(j.error ?? "ذخیره ناموفق بود.");
   265→      onSaved();
   266→      onClose();
   267→    } catch (e) {
   268→      setErr(e instanceof Error ? e.message : "خطایی رخ داد.");
   269→    } finally {
   270→      setBusy(false);
   271→    }
   272→  }
   273→
   274→  return (
   275→    <Modal open={!!draft} onClose={onClose} wide>
   276→      <div className="mb-4 flex items-center gap-2.5">
   277→        <span className="grid h-10 w-10 place-items-center rounded-xl bg-primary/10 text-primary"><PenSquare className="h-5 w-5" /></span>
   278→        <h2 className="font-display text-lg font-bold">{d.id ? "ویرایش مطلب" : "مطلب جدید"}</h2>
   279→        <button onClick={onClose} aria-label="بستن" className="ms-auto rounded-lg p-1.5 text-muted-foreground hover:bg-muted"><X className="h-5 w-5" /></button>
   280→      </div>
   281→
   282→      <div className="space-y-3">
   283→        <div>
   284→          <label className={labelCls}>عنوان مطلب *</label>
   285→          <input value={d.title} onChange={(e) => setD({ ...d, title: e.target.value })} placeholder="مثلاً: سه اشتباه رایج در قرارداد بیع" className={inputCls} />
   286→        </div>
   287→        <div className="grid gap-3 sm:grid-cols-2">
   288→          <div>
   289→            <label className={labelCls}>خلاصه (برای فید خانه)</label>
   290→            <input value={d.summary} onChange={(e) => setD({ ...d, summary: e.target.value })} maxLength={280} className={inputCls} />
   291→          </div>
   292→          <div>
   293→            <label className={labelCls}>برچسب‌ها (با کاما)</label>
   294→            <input value={d.tags} onChange={(e) => setD({ ...d, tags: e.target.value })} placeholder="قراردادها، بیع، نکته امتحانی" className={inputCls} />
   295→          </div>
   296→        </div>
   297→        <div>
   298→          <label className={labelCls}>محتوا</label>
   299→          <BlockEditor blocks={d.blocks} onChange={(blocks) => setD({ ...d, blocks })} />
   300→        </div>
   301→        {err && <p className="rounded-lg bg-destructive/10 px-3 py-2 text-sm text-destructive">{err}</p>}
   302→        <div className="flex justify-end gap-2 pt-1">
   303→          <button onClick={onClose} className="rounded-xl border border-border px-4 py-2.5 text-sm font-semibold text-muted-foreground hover:text-foreground">انصراف</button>
   304→          <button onClick={save} disabled={busy || !d.title.trim()} className="inline-flex items-center gap-2 rounded-xl bg-primary px-5 py-2.5 text-sm font-bold text-primary-foreground disabled:opacity-45">
   305→            {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : <Save className="h-4 w-4" />} انتشار / ذخیره
   306→          </button>
   307→        </div>
   308→      </div>
   309→    </Modal>
   310→  );
   311→}

   314→interface TLesson { key: string; title: string; minutes?: number; sections: LessonSection[] }
   315→interface TChapter { key: string; title: string; lessons: TLesson[] }
   316→interface CourseDraft { id?: string; title: string; tagline: string; description: string; icon: string; accent: string; chapters: TChapter[] }
   317→
   318→const EMPTY_COURSE: CourseDraft = { title: "", tagline: "", description: "", icon: "Scale", accent: "bronze", chapters: [] };
   319→const uid = () => Math.random().toString(36).slice(2, 9);

   321→export function CourseEditor({ draft, onClose, onSaved }: { draft: CourseDraft | null; onClose: () => void; onSaved: () => void }) {
   322→  const [d, setD] = React.useState<CourseDraft>(draft ?? EMPTY_COURSE);
   323→  const [busy, setBusy] = React.useState(false);
   324→  const [err, setErr] = React.useState("");
   325→  const [openCh, setOpenCh] = React.useState<string>("");
   326→
   327→  React.useEffect(() => {
   328→    setD(draft ?? EMPTY_COURSE);
   329→    setOpenCh((draft?.chapters ?? [])[0]?.key ?? "");
   330→  }, [draft]);
   331→
   332→  function mutate(fn: (x: CourseDraft) => CourseDraft) { setD(fn(d)); }
   333→
   334→  async function save() {
   335→    setBusy(true); setErr("");
   336→    try {
   337→      // اعتبارسنجی سرهٔ کلاینت: فصل خالی حذف نمی‌شود ولی جلسهٔ خالی رد می‌شود
   338→      const chapters = d.chapters.map((c) => ({ title: c.title, lessons: c.lessons.map((l) => ({ title: l.title, minutes: l.minutes, sections: l.sections })) }));
   339→      const res = await fetch("/api/tcourses", {
   340→        method: "POST",
   341→        headers: { "Content-Type": "application/json" },
   342→        body: JSON.stringify({ ...d, chapters }),
   343→      });
   344→      const j = await res.json();
   345→      if (!res.ok) throw new Error(j.error ?? "ذخیره ناموفق بود.");
   346→      onSaved();
   347→      onClose();
   348→    } catch (e) {
   349→      setErr(e instanceof Error ? e.message : "خطایی رخ داد.");
   350→    } finally {
   351→      setBusy(false);
   352→    }
   353→  }
   354→
   355→  return (
   356→    <Modal open={!!draft} onClose={onClose} wide>
   357→      <div className="mb-4 flex items-center gap-2.5">
   358→        <span className="grid h-10 w-10 place-items-center rounded-xl bg-bronze/15 text-bronze"><GraduationCap className="h-5 w-5" /></span>
   359→        <h2 className="font-display text-lg font-bold">{d.id ? "ویرایش دوره" : "دورهٔ جدید"}</h2>
   360→        <button onClick={onClose} aria-label="بستن" className="ms-auto rounded-lg p-1.5 text-muted-foreground hover:bg-muted"><X className="h-5 w-5" /></button>
   361→      </div>
   362→

   471→          </button>
