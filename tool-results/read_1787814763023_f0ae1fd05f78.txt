    55→/** عنوان جلسه از شناسه (برای نمایش خوانا) */
    56→function prettyLesson(id: string): string {
    57→  const m = id.match(/^(m|t|md7)-l(\d+)-(\d+)$/);
    58→  if (!m) return id;
    59→  const book = m[1] === "t" ? "تجارت ۳" : m[1] === "md7" ? "مدنی ۷" : `مدنی ${m[1] === "m" ? "" : ""}${m[2]}`;
    60→  return `${book} — جلسهٔ ${fa(m[3])} فصل ${fa(m[2])}`;
    61→}
    62→
    63→function scoreColor(s: number) {
    64→  return s >= 80 ? "text-success" : s >= 60 ? "text-bronze" : "text-destructive";
    65→}

    83→// ─── هیت‌استریپ فعالیت (۱۲ هفتهٔ اخیر) ────────────────────────────────────────
    84→function HeatStrip({ days }: { days: string[] }) {
    85→  const set = new Set(days);
    86→  const cells = Array.from({ length: 84 }, (_, i) => {
    87→    const d = new Date(Date.now() - (83 - i) * 86_400_000);
    88→    const iso = d.toISOString().slice(0, 10);
    89→    return set.has(iso);
    90→  });
    91→  return (
    92→    <div className="flex flex-wrap gap-[3px]" dir="ltr" aria-label={`فعال در ${fa(set.size)} روز`}>
    93→      {cells.map((hot, i) => (
    94→        <span
    95→          key={i}
    96→          aria-hidden
    97→          className={`h-2.5 w-2.5 rounded-[3px] ${hot ? "bg-bronze" : "bg-border/70"}`}
    98→          style={hot ? { opacity: 0.65 + Math.min(0.35, set.size / 120) } : undefined}
    99→        />
   100→      ))}
   101→    </div>
   102→  );
   103→}

   339→function AdminAccountCard() {
   340→  const auth = useAuth();
   341→  const [currentPassword, setCurrentPassword] = React.useState("");
   342→  const [newUsername, setNewUsername] = React.useState("");
   343→  const [newPassword, setNewPassword] = React.useState("");
   344→  const [msg, setMsg] = React.useState<{ ok: boolean; text: string } | null>(null);
   345→  const [busy, setBusy] = React.useState(false);
   346→
   347→  async function save(e: React.FormEvent) {
   348→    e.preventDefault();
   349→    setBusy(true);
   350→    setMsg(null);
   351→    try {
   352→      const res = await fetch("/api/admin/account", {
   353→        method: "POST",
   354→        headers: { "Content-Type": "application/json" },
   355→        body: JSON.stringify({
   356→          currentPassword,
   357→          newUsername: newUsername.trim() || undefined,
   358→          newPassword: newPassword || undefined,
   359→        }),
   360→      });
   361→      const data = (await res.json()) as { ok?: boolean; error?: string; newUsername?: string };
   362→      if (data.ok) {
   363→        setMsg({ ok: true, text: "اطلاعات حساب مدیر به‌روزرسانی شد." });
   364→        setCurrentPassword("");
   365→        setNewPassword("");
   366→        if (data.newUsername) {
   367→          setNewUsername("");
   368→          // تازه‌سازی نام کاربری در همین نشست
   369→          window.location.reload();
   370→        }
   371→      } else {
   372→        setMsg({ ok: false, text: data.error ?? "خطای نامشخص." });
   373→      }
   374→    } catch {
   375→      setMsg({ ok: false, text: "ارتباط برقرار نشد." });
   376→    }
   377→    setBusy(false);
   378→  }
   379→
   380→  const empty = !currentPassword || (!newUsername.trim() && !newPassword);
   381→
   382→  return (
   383→    <form onSubmit={save} className="space-y-3.5 rounded-2xl border border-bronze/30 bg-card p-5 shadow-card">
   384→      <div>
   385→        <h3 className="flex items-center gap-2 font-display text-sm font-bold">
   386→          <KeyRound className="h-4 w-4 text-bronze" /> اطلاعات ورود مدیر (یوزر و پسورد اختصاصی خودتان)
   387→        </h3>
   388→        <p className="mt-1 text-[11px] leading-relaxed text-muted-foreground">
   389→          رمز فعلی را برای تأیید وارد کنید؛ سپس نام کاربری یا رمز تازه را تعیین کنید.
   390→          تا زمانی که خودتان این‌جا عوضش کنید، همان اطلاعات اولیه معتبر است.
   391→        </p>
   392→      </div>
   393→
   394→      <div className="grid gap-3 sm:grid-cols-2">
   395→        <label className="space-y-1">
   396→          <span className="flex items-center gap-1.5 text-xs font-bold text-muted-foreground">
   397→            <Lock className="h-3 w-3" /> رمز فعلی
   398→          </span>
   399→          <input
   400→            type="password"
   401→            dir="ltr"
   402→            value={currentPassword}
   403→            onChange={(e) => setCurrentPassword(e.target.value)}
   404→            required
   405→            className="w-full rounded-xl border border-border bg-background px-3 py-2.5 text-sm outline-none focus:border-bronze"
   406→          />
   407→        </label>
   408→        <label className="space-y-1">
   409→          <span className="flex items-center gap-1.5 text-xs font-bold text-muted-foreground">
   410→            <UserRound className="h-3 w-3" /> نام کاربری جدید (اختیاری)
   411→          </span>
   412→          <input
   413→            dir="ltr"
   414→            value={newUsername}
   415→            onChange={(e) => setNewUsername(e.target.value)}
   416→            placeholder="بدون تغییر"
   417→            className="w-full rounded-xl border border-border bg-background px-3 py-2.5 text-sm outline-none placeholder:text-muted-foreground/50 focus:border-bronze"
   418→          />
   419→        </label>

   427→            value={newPassword}

   433→      </div>

   435→      {msg && (

   445→        </p>
   446→      )}
   447→
   448→      <button
   449→        type="submit"
   450→        disabled={busy || empty}

   456→    </form>
