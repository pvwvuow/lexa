   246→        (q) => q?.q && q.options?.length >= 3 && q.options.some((o) => o.key === q.answer) && q.explanation,

   252→    } catch (e) {
   253→      setGenErr(e instanceof Error ? e.message : "تولید تست ناموفق بود.");

   760→                >

   763→                {answeredCount < questions.length && idx === questions.length - 1 ? (

   768→                ) : (
   769→                  <button
   770→                    onClick={finish}

   787→/** گروه‌بندی درس‌های یک دامنه به شکل { دوره ← [فصل ← جلسات] } */
   788→function groupChapters(items: { course: Course; chapter: Chapter; lesson: Lesson }[]) {
   789→  const byCourse = new Map<Course, Map<Chapter, Lesson[]>>();
   790→  for (const f of items) {
   791→    if (!byCourse.has(f.course)) byCourse.set(f.course, new Map());
   792→    const ch = byCourse.get(f.course)!;
   793→    if (!ch.has(f.chapter)) ch.set(f.chapter, []);
   794→    ch.get(f.chapter)!.push(f.lesson);
   795→  }
   796→  return [...byCourse.entries()].map(([course, chMap]) => ({
   797→    course,
   798→    groups: [...chMap.entries()].map(([chapter, lessons]) => ({ chapter, lessons })),
   799→  }));
   800→}

   828→function shuffle<T>(arr: T[]): T[] {
   829→  const a = [...arr];
   830→  for (let i = a.length - 1; i > 0; i--) {
   831→    const j = Math.floor(Math.random() * (i + 1));
   832→    [a[i], a[j]] = [a[j], a[i]];
   833→  }
   834→  return a;
   835→}
