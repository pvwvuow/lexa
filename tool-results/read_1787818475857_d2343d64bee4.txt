   130→          chapterTitle: chapter.title,
   131→          lessonTitle: lesson.title,
   132→          seenSections: sections.slice(0, visibleCount).map((s) => s.title ?? ""),

   520→/** پاسخ استاد: متن Markdown + خطوط 📜 (مستند قانونی) در باکس‌های جدا */
   521→function AiAnswerRich({ text }: { text: string }) {
   522→  const { prose, laws } = React.useMemo(() => {
   523→    const lawLines: string[] = [];
   524→    const proseLines: string[] = [];
   525→    for (const ln of text.split("\n")) {
   526→      const t = ln.trim();
   527→      if (t.startsWith("📜")) lawLines.push(t.replace(/^📜\s*/, ""));
   528→      else proseLines.push(ln);
   529→    }
   530→    return { prose: proseLines.join("\n").replace(/\n{3,}/g, "\n\n").trim(), laws: lawLines };
   531→  }, [text]);
   532→  return (
   533→    <>
   534→      {prose && (
   535→        <div className="teach-body prose-p:leading-[1.9] text-[17px] [&_p]:my-1 [&_strong]:text-foreground">
   536→          <ReactMarkdown>{prose}</ReactMarkdown>
   537→        </div>
   538→      )}
   539→      {laws.length > 0 && (
   540→        <div className="mt-3 space-y-2">
   541→          <p className="text-[11px] font-bold text-muted-foreground">مستند قانونی پاسخ:</p>
   542→          {laws.map((l, i) => (
   543→            <p key={i} className="law-text flex gap-2 rounded-lg border-s-2 border-bronze/60 bg-background/70 px-3 py-2.5 text-[15.5px] leading-[1.9]">
   544→              <Scale className="mt-1.5 h-3.5 w-3.5 shrink-0 text-bronze" />
   545→              <span>{l}</span>
   546→            </p>
   547→          ))}
   548→        </div>
   549→      )}
   550→    </>
   551→  );
   552→}
