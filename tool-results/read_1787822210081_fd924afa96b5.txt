   122→/**
   123→ * گیج پیشرفت نسخهٔ ۲ — حلقهٔ گرادیانی یشمی→طلایی با تیک‌های محیطی
   124→ * (نام قبلی Donut حفظ شده تا جاهای دیگر نشکند)
   125→ */
   126→export function Donut({
   127→  value, size = 92, stroke = 9, label, flat,
   128→}: { value: number; size?: number; stroke?: number; label?: string; flat?: boolean }) {
   129→  const id = React.useId().replace(/[^a-zA-Z0-9]/g, "");
   130→  const r = (size - stroke - 8) / 2;
   131→  const c = 2 * Math.PI * r;
   132→  const v = Math.min(100, Math.max(0, value));
   133→  const filled = (v / 100) * c;
   134→
   135→  // تیک‌های ظریف محیط بیرونی
   136→  const ticks = Array.from({ length: 36 }, (_, i) => i * 10);
   137→
   138→  return (
   139→    <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} role="img" aria-label={`پیشرفت ${value} درصد`}>
   140→      {!flat && (
   141→        <defs>
   142→          <linearGradient id={`g-${id}`} x1="0%" y1="0%" x2="100%" y2="100%">
   143→            <stop offset="0%" stopColor="var(--primary)" />
   144→            <stop offset="100%" stopColor="var(--bronze)" />
   145→          </linearGradient>
   146→        </defs>
   147→      )}
   148→
   149→      {/* تیک‌های محیطی */}
   150→      <g opacity="0.22">
   151→        {ticks.map((deg) => (
   152→          <line
   153→            key={deg} x1={size / 2} y1={4} x2={size / 2} y2={Math.max(6, stroke > 9 ? 9 : 7)}
   154→            stroke="currentColor" strokeWidth="1" strokeLinecap="round"
   155→            transform={`rotate(${deg + 90} ${size / 2} ${size / 2})`}
   156→          />
   157→        ))}
   158→      </g>
   159→
   160→      {/* ریل */}
   161→      <circle cx={size / 2} cy={size / 2} r={r} fill="none" strokeWidth={stroke} className="text-border" stroke="currentColor" />
   162→      {/* مقدار */}
   163→      <circle
   164→        cx={size / 2} cy={size / 2} r={r} fill="none"
   165→        stroke={flat ? "var(--bronze)" : `url(#g-${id})`} strokeWidth={stroke} strokeLinecap="round"
   166→        strokeDasharray={`${filled} ${c}`} transform={`rotate(-90 ${size / 2} ${size / 2})`}
   167→        style={{ transition: "stroke-dasharray .7s cubic-bezier(.4,0,.2,1)" }}
   168→      />
   169→
   170→      <text x="50%" y="47%" textAnchor="middle" dominantBaseline="middle" className="fill-foreground font-display text-lg font-bold">
   171→        {fa(v)}٪
   172→      </text>
   173→      {label && (
   174→        <text x="50%" y="67%" textAnchor="middle" dominantBaseline="middle" className="fill-muted-foreground text-[10px]">
   175→          {label}
   176→        </text>
   177→      )}
   178→    </svg>
   179→  );
   180→}

   211→/* ═══ موتور محتوای غنی: تبدیل متن درس به بلوک‌های ساخت‌یافته ═══════════════ */
   212→
   213→type SubItem = { term?: string; text: string };
   214→type TermItem = { term?: string; text: string; subs?: SubItem[] };
   215→type BodyBlock =
   216→  | { kind: "p"; text: string }
   217→  | { kind: "terms"; items: TermItem[] }
   218→  | { kind: "steps"; items: string[] };
   219→
   220→const DASH_RE = /^[-–•*]\s+/;
   221→const NUM_RE = /^[0-9۰-۹]{1,2}\s*[-–.)))]\s*/;
   222→
   223→type RunKind = "p" | "dash" | "num";
   224→
   225→/** جداسازی عنوان و توضیح از خطوطی مثل «اهلیت تمتع: توان دارا شدن حق» */
   226→function parseTermLine(line: string): TermItem {
   227→  const raw = line.replace(/^[-–•*]\s+/, "").trim();
   228→  const m = raw.match(/^(«?[^«»:：]{2,44}»?)\s*[:：]\s+(.+)$/);
   229→  if (m) return { term: m[1].replace(/^«/, "").replace(/»$/, ""), text: m[2].trim() };
   230→  return { text: raw };
   231→}
   232→
   233→/** بدنهٔ درس را به پاراگراف / کارت اصطلاح / پله‌نما تبدیل می‌کند.
   234→ *  خط‌به‌خط گروه‌بندی می‌شود؛ هیچ جملهٔ مقدمه‌ای حذف نمی‌شود و
   235→ *  خطوط فرورفته به‌عنوان زیرمجموعهٔ آیتم قبلی رندر می‌شوند. */
   236→export function parseBody(body: string): BodyBlock[] {
   237→  const blocks: BodyBlock[] = [];
   238→  let paraBuf: string[] = [];
   239→  const flushPara = () => {
   240→    if (paraBuf.length) {
   241→      blocks.push({ kind: "p", text: paraBuf.join("\n") });
   242→      paraBuf = [];
   243→    }
   244→  };
   245→
   246→  for (const chunk of body.split(/\n{2,}/)) {
   247→    const rawLines = chunk.split("\n").filter((l) => l.trim());
   248→    if (!rawLines.length) continue;
   249→
   250→    // گروه‌بندی خط‌های پیوستهٔ هم‌نوع
   251→    const runs: { kind: RunKind; lines: string[] }[] = [];
   252→    for (const raw of rawLines) {
   253→      const t = raw.trim();
   254→      const kind: RunKind = DASH_RE.test(t) ? "dash" : NUM_RE.test(t) ? "num" : "p";
   255→      const lastRun = runs[runs.length - 1];
   256→      if (kind === "p") {
   257→        if (!lastRun || lastRun.kind !== "p") runs.push({ kind: "p", lines: [raw] });
   258→        else lastRun.lines.push(raw);
   259→      } else if (lastRun && lastRun.kind === kind) lastRun.lines.push(raw);
   260→      else runs.push({ kind, lines: [raw] });
   261→    }
   262→
   263→    for (const run of runs) {
   264→      if (run.kind === "p") {
   265→        paraBuf.push(run.lines.map((l) => l.trim()).join("\n"));
   266→        continue;
   267→      }
   268→      flushPara();
   269→      if (run.kind === "num") {
   270→        blocks.push({ kind: "steps", items: run.lines.map((l) => l.replace(NUM_RE, "").trim()) });
   271→      } else {
   272→        const items: TermItem[] = [];
   273→        for (const raw of run.lines) {
   274→          const indented = /^[ \t]/.test(raw);
   275→          const parsed = parseTermLine(raw.trim());
   276→          if (indented && items.length) {
   277→            const parent = items[items.length - 1];
   278→            (parent.subs ??= []).push({ term: parsed.term, text: parsed.text });
   279→          } else items.push(parsed);
   280→        }
   281→        blocks.push({ kind: "terms", items });
   282→      }
   283→    }
   284→  }
   285→  flushPara();
   286→  return blocks;
   287→}

   367→/* ── برگهٔ مرور — رندر اختصاصی جمع‌بندی ═══════════════════════════════════════
   368→ * به‌جای کارت‌های مستطیلیِ تکراری، حس «برگهٔ چکیدهٔ پرونده»: اعداد آویزان، متن
   369→ * روان و جداکنندهٔ مویی. حالت خودآزمایی هم دارد: ابتدا جمله پنهان است و با کلیک
   370→ * روی هر ردیف، پاسخ همان نکته باز می‌شود (یادآوری فعال). */
   371→export function SummarySheet({ items }: { items: string[] }) {
   372→  const parsed = React.useMemo(
   373→    () => items.map((x) => parseTermLine(x.replace(NUM_RE, "").trim())),
   374→    [items],
   375→  );
   376→  const [veil, setVeil] = React.useState(false);
   377→  const [shown, setShown] = React.useState<Set<number>>(new Set());
   378→  const revealed = shown;
   379→
   380→  React.useEffect(() => {
   381→    setShown(new Set());
   382→    setVeil(false);
   383→  }, [items]);
   384→
   385→  if (!parsed.length) return null;
   386→
   387→  const reveal = (i: number) => setShown((s) => { const n = new Set(s); n.add(i); return n; });
   388→
   389→  return (
   390→    <div>
   391→      {/* نوار ابزار برگه */}
   392→      <div className="mb-2 flex flex-wrap items-center justify-between gap-2">
   393→        <span className="text-[11.5px] font-medium text-muted-foreground">
   394→          چکیدهٔ جلسه در {fa(parsed.length)} نکته{veil && <> — {fa(revealed.size)} از {fa(parsed.length)} باز شده</>}
   395→        </span>
   396→        <button
   397→          onClick={() => { setVeil((v) => !v); setShown(new Set()); }}
   398→          aria-pressed={veil}
   399→          className={`inline-flex items-center gap-1.5 rounded-full border px-3 py-1 text-[11.5px] font-bold transition-colors ${
   400→            veil ? "border-bronze bg-bronze/15 text-bronze" : "border-border bg-background text-muted-foreground hover:border-bronze/50 hover:text-bronze"
   401→          }`}
   402→        >

   461→/** رندر بدنهٔ درس: پاراگراف + کارت اصطلاح + پله‌نما (تشخیص خودکار نکتهٔ مهم) */
   462→export function BodyRich({ text }: { text: string }) {
   463→  const blocks = React.useMemo(() => parseBody(text), [text]);
   464→  return (
   465→    <div className="space-y-4">
   466→      {blocks.map((b, i) => {
   467→        if (b.kind === "p") {
   468→          // تشخیص «نکتهٔ مهم» خط‌به‌خط — پاراگراف‌های معمولی و المان‌های مهم
   469→          // می‌توانند در یک بلوک کنار هم باشند؛ هر خط جداگانه ارزیابی می‌شود
   470→          const pieces = b.text.split("\n");
   471→          const rendered: React.ReactNode[] = [];
   472→          let buf: string[] = [];
   473→          const flush = (key: string) => {
   474→            if (buf.length)
   475→              rendered.push(
   476→                <p key={key} className="whitespace-pre-line text-[20px] leading-[2.15] text-foreground/95">
   477→                  {buf.join("\n")}
   478→                </p>,
   479→              );
   480→            buf = [];
   481→          };
   482→          pieces.forEach((ln, k) => {
   483→            const imp = detectImportant(ln);
   484→            if (imp) {
   485→              flush(`p-${i}-${k}`);
   486→              rendered.push(
   487→                <ImportantNote key={`i-${i}-${k}`} variant={imp.variant} label={imp.label}>
   488→                  {imp.rest}
   489→                </ImportantNote>,
   490→              );
   491→            } else buf.push(ln);
   492→          });
   493→          flush(`p-${i}-end`);
   494→          if (rendered.length === 1 && React.isValidElement(rendered[0])) return rendered[0];
   495→          return <React.Fragment key={i}>{rendered}</React.Fragment>;
   496→        }
   497→        if (b.kind === "steps") return <StepList key={i} items={b.items} />;
   498→        const withTerm = b.items.filter((x) => x.term || x.subs).length >= Math.ceil(b.items.length / 2);
   499→        if (!withTerm) return <PlainList key={i} items={b.items.map((x) => x.text)} />;
   500→        return (
   501→          <div key={i} className="grid items-start gap-3 md:grid-cols-2">
   502→            {b.items.map((x, j) => <TermCard key={j} item={x} index={b.items.length > 1 && (x.term || x.subs) ? j + 1 : undefined} />)}
   503→          </div>
   504→        );
   505→      })}
   506→    </div>
   507→  );
   508→}

   576→/* ── المان «نکتهٔ مهم» در دل متن ══════════════════════════════════════════════
   577→ * باند تمام‌عرض بدون قاب مستطیلی: میلهٔ گردِ رنگی + لوزی آیکون + پس‌زمینهٔ
   578→ * محو رنگی متناسب با شدت پیام (هشدار سرخ، مهم کهربا، ترفند یشمی، یادآوری سرمه‌ای). */
   579→
   580→type ImpVariant = "warn" | "danger" | "success" | "info";
   581→
   582→type ImpIcon = React.ComponentType<{ className?: string; style?: React.CSSProperties }>;
   583→const IMP_STYLE: Record<ImpVariant, { colorVar: string; icon: ImpIcon; fallbackLabel: string }> = {
   584→  warn: { colorVar: "--warn", icon: AlertTriangle, fallbackLabel: "نکتهٔ مهم" },
   585→  danger: { colorVar: "--destructive", icon: AlertTriangle, fallbackLabel: "هشدار" },
   586→  success: { colorVar: "--success", icon: Zap, fallbackLabel: "ترفند" },
   587→  info: { colorVar: "--primary", icon: Info, fallbackLabel: "یادآوری" },
   588→};
   589→
   590→/** تشخیص پاراگراف‌های «نکتهٔ مهم» از سرنخ متن — برای reuse در رندررها.
   591→ *  هم «هشدار: …» و هم برچسب‌های بلندتر مثل «هشدار اهلیت در عاریه: …» را می‌گیرد. */
   592→export function detectImportant(text: string): { variant: ImpVariant; label: string; rest: string } | null {
   593→  const m = text.match(
   594→    /^\s*(⚠️|❗|📌|🚨)?\s*\[?\s*((هشدار|توجه مهم|توجه|مهم|نکتهٔ مهم|نکته مهم|یادآوری|ترفند)[^:：\n]{0,42}?)\s*\]?\s*[:：]\s*/,
   595→  );
   596→  if (!m) return null;
   597→  const kw = m[3];
   598→  const fullLabel = m[2].trim();
   599→  let variant: ImpVariant = "warn";
   600→  if (/هشدار/.test(kw)) variant = "danger";
   601→  else if (/ترفند/.test(kw)) variant = "success";
   602→  else if (/یادآوری/.test(kw)) variant = "info";
   603→  else variant = "warn";
   604→  return { variant, label: fullLabel, rest: text.slice(m[0].length).trim() };
   605→}
