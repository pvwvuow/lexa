   147→/**
   148→ * گیج پیشرفت نسخهٔ ۲ — حلقهٔ گرادیانی یشمی→طلایی با تیک‌های محیطی
   149→ * (نام قبلی Donut حفظ شده تا جاهای دیگر نشکند)
   150→ */
   151→export function Donut({
   152→  value, size = 92, stroke = 9, label, flat,
   153→  /** برای سطوح تیره (مثل هیرو) — رنگ متن درصد و برچسب */
   154→  toneClass = "fill-foreground",
   155→  labelToneClass = "fill-muted-foreground",
   156→}: { value: number; size?: number; stroke?: number; label?: string; flat?: boolean; toneClass?: string; labelToneClass?: string }) {
   157→  const id = React.useId().replace(/[^a-zA-Z0-9]/g, "");
   158→  const r = (size - stroke - 8) / 2;
   159→  const c = 2 * Math.PI * r;
   160→  const v = Math.min(100, Math.max(0, value));
   161→  const filled = (v / 100) * c;
   162→
   163→  // تیک‌های ظریف محیط بیرونی
   164→  const ticks = Array.from({ length: 36 }, (_, i) => i * 10);
   165→
   166→  return (
   167→    <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} role="img" aria-label={`پیشرفت ${value} درصد`}>
   168→      {!flat && (
   169→        <defs>
   170→          <linearGradient id={`g-${id}`} x1="0%" y1="0%" x2="100%" y2="100%">
   171→            <stop offset="0%" stopColor="var(--primary)" />
   172→            <stop offset="100%" stopColor="var(--bronze)" />
   173→          </linearGradient>
   174→        </defs>
   175→      )}
   176→
   177→      {/* تیک‌های محیطی */}
   178→      <g opacity="0.22">
   179→        {ticks.map((deg) => (
   180→          <line
   181→            key={deg} x1={size / 2} y1={4} x2={size / 2} y2={Math.max(6, stroke > 9 ? 9 : 7)}
   182→            stroke="currentColor" strokeWidth="1" strokeLinecap="round"
   183→            transform={`rotate(${deg + 90} ${size / 2} ${size / 2})`}
   184→          />
   185→        ))}
   186→      </g>
   187→
   188→      {/* ریل */}
   189→      <circle cx={size / 2} cy={size / 2} r={r} fill="none" strokeWidth={stroke} className="text-border" stroke="currentColor" />
   190→      {/* مقدار */}
   191→      <circle
   192→        cx={size / 2} cy={size / 2} r={r} fill="none"
   193→        stroke={flat ? "var(--bronze)" : `url(#g-${id})`} strokeWidth={stroke} strokeLinecap="round"
   194→        strokeDasharray={`${filled} ${c}`} transform={`rotate(-90 ${size / 2} ${size / 2})`}
   195→        style={{ transition: "stroke-dasharray .7s cubic-bezier(.4,0,.2,1)" }}
   196→      />
   197→
   198→      <text x="50%" y="47%" textAnchor="middle" dominantBaseline="middle" className={toneClass + " font-display text-lg font-bold"}>
   199→        {fa(v)}٪
   200→      </text>
   201→      {label && (
   202→        <text x="50%" y="67%" textAnchor="middle" dominantBaseline="middle" className={labelToneClass + " text-[10px]"}>
   203→          {label}
   204→        </text>
   205→      )}
   206→    </svg>
   207→  );
   208→}

   239→/* ═══ موتور محتوای غنی: تبدیل متن درس به بلوک‌های ساخت‌یافته ═══════════════ */
   240→
   241→type SubItem = { term?: string; text: string };
   242→type TermItem = { term?: string; text: string; subs?: SubItem[] };
   243→type BodyBlock =
   244→  | { kind: "p"; text: string }
   245→  | { kind: "terms"; items: TermItem[] }
   246→  | { kind: "steps"; items: string[] };
   247→
   248→const DASH_RE = /^[-–•*]\s+/;
   249→const NUM_RE = /^[0-9۰-۹]{1,2}\s*[-–.)))]\s*/;
   250→
   251→type RunKind = "p" | "dash" | "num";
   252→
   253→/** جداسازی عنوان و توضیح از خطوطی مثل «اهلیت تمتع: توان دارا شدن حق» */
   254→function parseTermLine(line: string): TermItem {
   255→  const raw = line.replace(/^[-–•*]\s+/, "").trim();
   256→  const m = raw.match(/^(«?[^«»:：]{2,44}»?)\s*[:：]\s+(.+)$/);
   257→  if (m) return { term: m[1].replace(/^«/, "").replace(/»$/, ""), text: m[2].trim() };
   258→  return { text: raw };
   259→}
   260→
   261→/** بدنهٔ درس را به پاراگراف / کارت اصطلاح / پله‌نما تبدیل می‌کند.
   262→ *  خط‌به‌خط گروه‌بندی می‌شود؛ هیچ جملهٔ مقدمه‌ای حذف نمی‌شود و
   263→ *  خطوط فرورفته به‌عنوان زیرمجموعهٔ آیتم قبلی رندر می‌شوند. */
   264→export function parseBody(body: string): BodyBlock[] {
   265→  const blocks: BodyBlock[] = [];
   266→  let paraBuf: string[] = [];
   267→  const flushPara = () => {
   268→    if (paraBuf.length) {
   269→      blocks.push({ kind: "p", text: paraBuf.join("\n") });
   270→      paraBuf = [];
   271→    }
   272→  };
   273→
   274→  for (const chunk of body.split(/\n{2,}/)) {
   275→    const rawLines = chunk.split("\n").filter((l) => l.trim());
   276→    if (!rawLines.length) continue;
   277→
   278→    // گروه‌بندی خط‌های پیوستهٔ هم‌نوع
   279→    const runs: { kind: RunKind; lines: string[] }[] = [];
   280→    for (const raw of rawLines) {
   281→      const t = raw.trim();
   282→      const kind: RunKind = DASH_RE.test(t) ? "dash" : NUM_RE.test(t) ? "num" : "p";
   283→      const lastRun = runs[runs.length - 1];
   284→      if (kind === "p") {
   285→        if (!lastRun || lastRun.kind !== "p") runs.push({ kind: "p", lines: [raw] });
   286→        else lastRun.lines.push(raw);
   287→      } else if (lastRun && lastRun.kind === kind) lastRun.lines.push(raw);
   288→      else runs.push({ kind, lines: [raw] });
   289→    }
   290→
   291→    for (const run of runs) {
   292→      if (run.kind === "p") {
   293→        paraBuf.push(run.lines.map((l) => l.trim()).join("\n"));
   294→        continue;
   295→      }
   296→      flushPara();
   297→      if (run.kind === "num") {
   298→        blocks.push({ kind: "steps", items: run.lines.map((l) => l.replace(NUM_RE, "").trim()) });
   299→      } else {
   300→        const items: TermItem[] = [];
   301→        for (const raw of run.lines) {
   302→          const indented = /^[ \t]/.test(raw);
   303→          const parsed = parseTermLine(raw.trim());
   304→          if (indented && items.length) {
   305→            const parent = items[items.length - 1];
   306→            (parent.subs ??= []).push({ term: parsed.term, text: parsed.text });
   307→          } else items.push(parsed);
   308→        }
   309→        blocks.push({ kind: "terms", items });
   310→      }
   311→    }
   312→  }
   313→  flushPara();
   314→  return blocks;
   315→}

   397→ * روان و جداکنندهٔ مویی. حالت خودآزمایی هم دارد: ابتدا جمله پنهان است و با کلیک

   489→/** رندر بدنهٔ درس: پاراگراف + کارت اصطلاح + پله‌نما (تشخیص خودکار نکتهٔ مهم) */
   490→export function BodyRich({ text }: { text: string }) {
   491→  const blocks = React.useMemo(() => parseBody(text), [text]);
   492→  return (
   493→    <div className="space-y-4">
   494→      {blocks.map((b, i) => {
   495→        if (b.kind === "p") {
   496→          // تشخیص «نکتهٔ مهم» خط‌به‌خط — پاراگراف‌های معمولی و المان‌های مهم
   497→          // می‌توانند در یک بلوک کنار هم باشند؛ هر خط جداگانه ارزیابی می‌شود
   498→          const pieces = b.text.split("\n");
   499→          const rendered: React.ReactNode[] = [];
   500→          let buf: string[] = [];
   501→          const flush = (key: string) => {
   502→            if (buf.length)
   503→              rendered.push(
   504→                <p key={key} className="whitespace-pre-line text-[20px] leading-[2.15] text-foreground/95">
   505→                  {buf.join("\n")}
   506→                </p>,
   507→              );
   508→            buf = [];
   509→          };
   510→          pieces.forEach((ln, k) => {
   511→            const imp = detectImportant(ln);
   512→            if (imp) {
   513→              flush(`p-${i}-${k}`);
   514→              rendered.push(
   515→                <ImportantNote key={`i-${i}-${k}`} variant={imp.variant} label={imp.label}>
   516→                  {imp.rest}
   517→                </ImportantNote>,
   518→              );
   519→            } else buf.push(ln);
   520→          });
   521→          flush(`p-${i}-end`);
   522→          if (rendered.length === 1 && React.isValidElement(rendered[0])) return rendered[0];
   523→          return <React.Fragment key={i}>{rendered}</React.Fragment>;
   524→        }
   525→        if (b.kind === "steps") return <StepList key={i} items={b.items} />;
   526→        const withTerm = b.items.filter((x) => x.term || x.subs).length >= Math.ceil(b.items.length / 2);
   527→        if (!withTerm) return <PlainList key={i} items={b.items.map((x) => x.text)} />;
   528→        return (
   529→          <div key={i} className="grid items-start gap-3 md:grid-cols-2">
   530→            {b.items.map((x, j) => <TermCard key={j} item={x} index={b.items.length > 1 && (x.term || x.subs) ? j + 1 : undefined} />)}
   531→          </div>
   532→        );
   533→      })}
   534→    </div>
   535→  );

   606→ * محو رنگی متناسب با شدت پیام (هشدار سرخ، مهم کهربا، ترفند یشمی، یادآوری سرمه‌ای). */
   607→
   608→type ImpVariant = "warn" | "danger" | "success" | "info";
   609→
   610→type ImpIcon = React.ComponentType<{ className?: string; style?: React.CSSProperties }>;
   611→const IMP_STYLE: Record<ImpVariant, { colorVar: string; icon: ImpIcon; fallbackLabel: string }> = {
   612→  warn: { colorVar: "--warn", icon: AlertTriangle, fallbackLabel: "نکتهٔ مهم" },
   613→  danger: { colorVar: "--destructive", icon: AlertTriangle, fallbackLabel: "هشدار" },
   614→  success: { colorVar: "--success", icon: Zap, fallbackLabel: "ترفند" },
   615→  info: { colorVar: "--primary", icon: Info, fallbackLabel: "یادآوری" },

   618→/** تشخیص پاراگراف‌های «نکتهٔ مهم» از سرنخ متن — برای reuse در رندررها.
   619→ *  هم «هشدار: …» و هم برچسب‌های بلندتر مثل «هشدار اهلیت در عاریه: …» را می‌گیرد. */
   620→export function detectImportant(text: string): { variant: ImpVariant; label: string; rest: string } | null {
   621→  const m = text.match(
   622→    /^\s*(⚠️|❗|📌|🚨)?\s*\[?\s*((هشدار|توجه مهم|توجه|مهم|نکتهٔ مهم|نکته مهم|یادآوری|ترفند)[^:：\n]{0,42}?)\s*\]?\s*[:：]\s*/,
   623→  );
   624→  if (!m) return null;
   625→  const kw = m[3];
   626→  const fullLabel = m[2].trim();
   627→  let variant: ImpVariant = "warn";
   628→  if (/هشدار/.test(kw)) variant = "danger";
   629→  else if (/ترفند/.test(kw)) variant = "success";
   630→  else if (/یادآوری/.test(kw)) variant = "info";
   631→  else variant = "warn";
   632→  return { variant, label: fullLabel, rest: text.slice(m[0].length).trim() };
   633→}
