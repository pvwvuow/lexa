    14→/* ─── نرمال‌سازی متنی فارسی برای جستجو ─────────────────────────────────── */
    15→function norm(s: string): string {
    16→  return s
    17→    .replace(/[ىي]/g, "ی")
    18→    .replace(/ك/g, "ک")
    19→    .replace(/[\u064B-\u0652\u200c]/g, "")
    20→    .replace(/[\u06F0-\u06F9\u0660-\u0669]/g, (d) => String(d.charCodeAt(0) & 0xf))
    21→    .toLowerCase();
    22→}

    50→/** ساخت شاخص سبک از کل کتابخانه — یک‌بار به ازای تغییر آرایهٔ دوره‌ها */
    51→function buildIndex(courses: Course[]) {
    52→  const items: { c: Course; ch: string; l: { id: string; title: string }; flatIdx: number; total: number; hay: string; plain: string }[] = [];
    53→  for (const c of courses) {
    54→    const flats = flatLessons(c);
    55→    const total = flats.length;
    56→    flats.forEach(({ lesson, chapter }, i) => {
    57→      let plain = `${lesson.title} ${chapter.title} ${c.title}`;
    58→      for (const s of lesson.sections) {
    59→        if (s.title) plain += ` ${s.title}`;
    60→        if (s.body) plain += ` ${s.body}`;
    61→        if (s.bullets) plain += ` ${s.bullets.join(" ")}`;
    62→        if (s.law) for (const w of s.law) plain += ` ماده ${w.no} ${w.text}`;
    63→        if (s.table) for (const r of s.table.rows) plain += ` ${r.join(" ")}`;
    64→        if (s.questionText) plain += ` ${s.questionText}`;
    65→      }
    66→      for (const q of lesson.quiz.slice(0, 3)) plain += ` ${q.q}`;
    67→      plain = plain.replace(/\s+/g, " ");
    68→      items.push({ c, ch: chapter.title, l: { id: lesson.id, title: lesson.title }, flatIdx: i, total, hay: norm(plain), plain });
    69→    });
    70→  }
    71→  return items;
    72→}

    87→/** شاخص سبک قوانین — یک‌بار ساخته می‌شود و همیشه در حافظه می‌ماند؛ با متن کامل غنی‌تر می‌شود */
    88→interface LawIndexItem { law: LawCode; no: string; text: string; book: string; chapter: string; hay: string; plain: string }
    89→let LAW_INDEX: LawIndexItem[] | null = null;
    90→function buildLawIndex(full?: Record<string, { books?: LawBook[] }> | null): LawIndexItem[] {
    91→  const items: LawIndexItem[] = [];
    92→  for (const law of LAW_CODES) {
    93→    const fbooks = full?.[law.id]?.books;
    94→    const books = fbooks?.length ? fbooks : law.books;
    95→    const word = law.articleWord === "اصل" ? "اصل" : "ماده";
    96→    for (const b of books)
    97→      for (const ch of b.chapters)
    98→        for (const a of ch.articles) {
    99→          const plain = `${a.text}`.replace(/\s+/g, " ");
   100→          items.push({ law, no: a.no, text: a.text, book: b.title, chapter: ch.title, hay: norm(`${law.title} ${word} ${a.no} ${plain}`), plain });
   101→        }
   102→  }
   103→  return items;
   104→}
